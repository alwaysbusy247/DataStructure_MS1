    static class InventoryItem implements Comparable<InventoryItem> {
        String dateEntered;
        String stockLabel;
        String brand;
        String engineNumber;
        String status;

        public InventoryItem(String dateEntered, String stockLabel,
                             String brand, String engineNumber,
                             String status) {
            this.dateEntered = dateEntered;
            this.stockLabel = stockLabel;
            this.brand = brand;
            this.engineNumber = engineNumber;
            this.status = status;
        }

        @Override
        public int compareTo(InventoryItem other) {
            return this.brand.compareTo(other.brand); // sort by brand
        }

        @Override
        public String toString() {
            return "Label: " + stockLabel +
                   ", Brand: " + brand +
                   ", Engine No: " + engineNumber +
                   ", Status: " + status;
        }
    }

    // Save inventory to Notepad (simple text file in Downloads)
    public static void saveToText(InventoryItem[] items) {
    String filename = "C:\\Users\\jhuns\\Downloads\\inventory.txt"; //change the name ex. jhuns to your username to access the file
    try (FileWriter writer = new FileWriter(filename)) {
        writer.write("MotorPH Inventory Records\n");
        writer.write("==========================\n");
        Arrays.sort(items); // <-- sort before saving
        for (InventoryItem item : items) {
            writer.write(item.toString() + "\n");
        }
        System.out.println("Inventory saved to " + filename);
    } catch (IOException e) {
        System.out.println("Error saving file: " + e.getMessage());
    }
}


    // Add new stock
    public static InventoryItem[] addItem(InventoryItem[] items, InventoryItem newItem) {
        InventoryItem[] newArray = Arrays.copyOf(items, items.length + 1);
        newArray[items.length] = newItem;
        return newArray;
    }

    // Delete stock by engine number
    public static InventoryItem[] deleteItem(InventoryItem[] items, String engineNumber) {
        InventoryItem[] temp = new InventoryItem[items.length - 1];
        int index = 0;
        boolean found = false;

        for (InventoryItem item : items) {
            if (item.engineNumber.equals(engineNumber)) {
                found = true;
            } else {
                if (index < temp.length) {
                    temp[index++] = item;
                }
            }
        }

        if (!found) {
            System.out.println("Stock not found.");
            return items;
        }

        System.out.println("Stock deleted successfully.");
        return Arrays.copyOf(temp, index);
    }

    // Search stock by engine number
    public static InventoryItem searchItem(InventoryItem[] items, String engineNumber) {
        for (InventoryItem item : items) {
            if (item.engineNumber.equals(engineNumber)) {
                return item;
            }
        }
        return null;
    }

    // Display (sorted by brand)
    public static void displayItems(InventoryItem[] items) {
        Arrays.sort(items);
        System.out.println("Stocks sorted by brand.");
        for (InventoryItem item : items) {
            System.out.println(item);
        }
    }
